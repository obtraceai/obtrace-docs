## Install

```bash
go get github.com/obtraceai/obtrace-sdk-go
```

## Configuration

```go
client := obtrace.NewClient(obtrace.Config{
  APIKey: os.Getenv("OBTRACE_API_KEY"),
  IngestBaseURL: os.Getenv("OBTRACE_INGEST_BASE_URL"),
  TenantID: os.Getenv("OBTRACE_TENANT_ID"),
  ProjectID: os.Getenv("OBTRACE_PROJECT_ID"),
  ServiceName: "go-api",
  AppID: "go-api",
  ValidateSemanticMetrics: true,
  Debug: true,
})
```

## Quick start

```go
client.Log("INFO", "service.started", nil)
client.Metric(obtrace.SemanticMetrics.RuntimeCPUUtilization, 0.42, "1", nil)
client.Span("db.query", "", "", 0, "", map[string]any{
  "db.system": "postgresql",
  "feature.name": "checkout",
})
_ = client.Flush(context.Background())
```
